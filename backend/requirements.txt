# Runtime dependencies for the backend API.
# Versions are pinned to the exact wheels verified on this machine
# (CPython 3.14 / Windows). Update deliberately, never implicitly.

# --- HTTP layer ---
fastapi==0.121.2
uvicorn==0.38.0
pydantic==2.12.4
pydantic-settings==2.16.2

# --- Observability ---
structlog==26.1.0
python-dotenv==1.2.4

# --- Persistence (F2) ---
# PyMySQL is the pure-Python driver for MySQL 8.x. `cryptography` is not
# optional: MySQL 8.4 defaults to the caching_sha2_password plugin, and PyMySQL
# refuses to authenticate with it unless cryptography is importable.
SQLAlchemy==2.0.55
PyMySQL==1.2.3
cryptography==50.0.2

# Alembic ships in the runtime image (not just dev) because migrations are
# applied by the deployment, not by a developer's laptop.
alembic==1.20.1

# --- Authentication (F3) ---
# argon2-cffi provides the Argon2id password hash `users.password_hash` stores.
# Its binding layer publishes an `abi3` wheel, so it works on 3.14 without a
# version-specific build. PyJWT is the HS256 token codec; it ships `py.typed`,
# so mypy strict analyses it rather than treating it as `Any`.
argon2-cffi==25.1.0
PyJWT==2.10.1

# --- Time zones (F10) ---
# tzdata supplies the IANA database that `zoneinfo` reads. It is not optional:
# a Windows image ships no system zone files, so `ZoneInfo("Europe/London")`
# raises `ZoneInfoNotFoundError` there and the quiet-hours check in the dispatch
# policy cannot resolve the contract's own zone. Pure Python, data only, no
# compiled extension. It is pinned like any other dependency so a silent tz
# refresh cannot change when a customer is called.
tzdata==2026.6

# --- Event transport (F4) ---
# kafka-python is the outbox relay's transport. It is pure Python (no compiled
# extension), so it installs on CPython 3.14 without a wheel hunt, and its
# send().get() future gives a real per-record broker acknowledgement — the
# relay treats "returned normally" as "delivered", so a fire-and-forget producer
# would not be a substitute. Imported lazily, so EVENT_TRANSPORT=none never
# needs it at runtime.
kafka-python==2.2.15

# --- Document text (gap 12) ---
# pypdf turns a stored contract PDF back into text. It is the pure-Python
# implementation (no compiled extension, so it installs on CPython 3.14 without a
# wheel hunt) and it ships `py.typed`, which is why mypy strict can analyse the
# call site in app/core/document_text.py instead of treating the parser as `Any`.
# Imported lazily there: a deployment that never parses a PDF does not pay for the
# import, and a process without this wheel refuses with a sentence naming the fix
# rather than failing at start-up. Chosen over a PDF-to-image or a service-based
# reader because the text layer is what the extractors need — rendering pages would
# add a heavyweight dependency for a document that usually already carries text.
pypdf==7.0.0
